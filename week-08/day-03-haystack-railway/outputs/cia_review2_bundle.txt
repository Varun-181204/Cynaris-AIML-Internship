import sys
from contextlib import asynccontextmanager
from pathlib import Path

from fastapi import FastAPI, HTTPException
from pydantic import BaseModel, Field

BASE_DIR = Path(__file__).resolve().parent
sys.path.insert(0, str(BASE_DIR / "src"))

from haystack_pipeline import HaystackRetrievalSystem

system = HaystackRetrievalSystem()


@asynccontextmanager
async def lifespan(app: FastAPI):
    system.index_pdfs()
    system.build_retrieval_pipelines()
    app.state.ready = True
    yield
    app.state.ready = False


app = FastAPI(
    title="Cynaris Haystack Retrieval API",
    description="BM25 and dense retrieval over five indexed PDF documents.",
    version="1.0.0",
    lifespan=lifespan,
)


class QueryRequest(BaseModel):
    question: str = Field(min_length=3, max_length=1000)
    method: str = Field(default="dense", pattern="^(bm25|dense)$")


@app.get("/health")
def health():
    return {
        "status": "ok",
        "ready": getattr(app.state, "ready", False),
    }


@app.post("/query")
def query(request: QueryRequest):
    if not getattr(app.state, "ready", False):
        raise HTTPException(status_code=503, detail="Pipeline is not ready.")

    try:
        result = system.run_query(request.method, request.question)
        return {
            "question": request.question,
            "method": request.method,
            "answer": result["reader"]["answer"],
            "source": result["reader"]["source"],
            "retrieved_sources": [
                doc.meta.get("source_name", "Unknown") for doc in result["documents"]
            ],
            "latency_ms": result["latency_ms"],
        }
    except Exception as exc:
        raise HTTPException(status_code=500, detail=str(exc)) from exc

@component
class ExtractiveReader:
    """Extract a query-relevant sentence from the highest-ranked document."""

    @component.output_types(answer=str, source=str, documents=list[Document])
    def run(self, query: str, documents: list[Document]):
        if not documents:
            return {
                "answer": "No relevant document was retrieved.",
                "source": "None",
                "documents": [],
            }

        query_terms = set(re.findall(r"\b\w+\b", query.lower()))
        best_document = documents[0]

        sentences = re.split(
            r"(?<=[.!?])\s+",
            best_document.content or "",
        )

        def sentence_score(sentence):
            terms = set(re.findall(r"\b\w+\b", sentence.lower()))
            return len(query_terms.intersection(terms))

        best_sentence = max(sentences, key=sentence_score, default="")

        return {
            "answer": best_sentence.strip() or "No readable answer found.",
            "source": best_document.meta.get("source_name", "Unknown"),
            "documents": documents,
        }


class HaystackRetrievalSystem:
    """Build and evaluate BM25 and dense Haystack retrieval."""

    def __init__(self):
        self.document_store = InMemoryDocumentStore(
            embedding_similarity_function="cosine"
        )

        self.indexing_pipeline = None
        self.bm25_pipeline = None
        self.dense_pipeline = None
        self.indexed_documents = 0

    def build_indexing_pipeline(self):
        converter = PyPDFToDocument()

        splitter = DocumentSplitter(
            split_by="word",
            split_length=180,
            split_overlap=30,
        )

        embedder = OllamaDocumentEmbedder(
            model=EMBEDDING_MODEL,
            url=OLLAMA_URL,
        )

        writer = DocumentWriter(
            document_store=self.document_store,
            policy=DuplicatePolicy.OVERWRITE,
        )

        pipeline = Pipeline()

        pipeline.add_component("file_converter", converter)
        pipeline.add_component("splitter", splitter)
        pipeline.add_component("embedder", embedder)
        pipeline.add_component("document_writer", writer)

        pipeline.connect("file_converter.documents", "splitter.documents")
        pipeline.connect("splitter.documents", "embedder.documents")
        pipeline.connect("embedder.documents", "document_writer.documents")

        self.indexing_pipeline = pipeline
        return pipeline

    def index_pdfs(self):
        pdf_files = sorted(PDF_DIR.glob("*.pdf"))

        if len(pdf_files) != 5:
            raise ValueError(f"Expected exactly 5 PDFs, found {len(pdf_files)}.")

        pipeline = self.build_indexing_pipeline()

        print("\nStarting PDF conversion and indexing...")
        print("PDF files:", len(pdf_files))

        result = pipeline.run(
            {
                "file_converter": {
                    "sources": pdf_files,
                    "meta": [{"source_name": pdf.name} for pdf in pdf_files],
                }
            }
        )

        self.indexed_documents = result["document_writer"]["documents_written"]

        print("Indexed document chunks:", self.indexed_documents)

        if self.indexed_documents == 0:
            raise RuntimeError("No document chunks were indexed.")

        return result

    def build_retrieval_pipelines(self):
        bm25_retriever = InMemoryBM25Retriever(
            document_store=self.document_store,
            top_k=TOP_K,
        )

        dense_retriever = InMemoryEmbeddingRetriever(
            document_store=self.document_store,
            top_k=TOP_K,
        )

        text_embedder = OllamaTextEmbedder(
            model=EMBEDDING_MODEL,
            url=OLLAMA_URL,
        )

        # BM25 retrieval and Reader
        bm25_pipeline = Pipeline()
        bm25_pipeline.add_component("retriever", bm25_retriever)
        bm25_pipeline.add_component("reader", ExtractiveReader())
        bm25_pipeline.connect("retriever.documents", "reader.documents")

        # Dense retrieval and Reader
        dense_pipeline = Pipeline()
        dense_pipeline.add_component("text_embedder", text_embedder)
        dense_pipeline.add_component("retriever", dense_retriever)
        dense_pipeline.add_component("reader", ExtractiveReader())

        dense_pipeline.connect(
            "text_embedder.embedding",
            "retriever.query_embedding",
        )
        dense_pipeline.connect(
            "retriever.documents",
            "reader.documents",
        )

        self.bm25_pipeline = bm25_pipeline
        self.dense_pipeline = dense_pipeline

    @staticmethod
    def unique_source_results(documents):
        """Keep the first highest-ranked chunk from each distinct PDF source."""
        unique_documents = []
        seen_sources = set()

        for document in documents:
            source = document.meta.get("source_name", "Unknown")
            if source not in seen_sources:
                seen_sources.add(source)
                unique_documents.append(document)

            if len(unique_documents) == TOP_K:
                break

        return unique_documents

    def run_query(self, method, question):
        start = time.perf_counter()

        if method == "bm25":
            result = self.bm25_pipeline.run(
                {
                    "retriever": {"query": question},
                    "reader": {"query": question},
                }
            )

        elif method == "dense":
            result = self.dense_pipeline.run(
                {
                    "text_embedder": {"text": question},
                    "reader": {"query": question},
                }
            )

        else:
            raise ValueError(f"Unknown retrieval method: {method}")

        elapsed_ms = (time.perf_counter() - start) * 1000

        documents = result["reader"]["documents"]
        unique_documents = self.unique_source_results(documents)

        return {
            "documents": unique_documents,
            "reader": result["reader"],
            "latency_ms": round(elapsed_ms, 2),
        }

    def evaluate(self):
        all_results = {}

        # Warm up Ollama before measuring dense-query latency.
        self.run_query("dense", QUESTIONS[0]["question"])

        for method in ("bm25", "dense"):
            print(f"\nEvaluating {method.upper()} retrieval...")

            method_results = []
            total_precision = 0.0

            for item in QUESTIONS:
                result = self.run_query(
                    method,
                    item["question"],
                )

                retrieved = result["documents"]
                retrieved_sources = [
                    doc.meta.get("source_name", "Unknown") for doc in retrieved
                ]

                relevant = set(item["relevant_sources"])

                relevant_hits = sum(source in relevant for source in retrieved_sources)

                precision = relevant_hits / TOP_K
                total_precision += precision

                record = {
                    "question_id": item["id"],
                    "question": item["question"],
                    "expected_sources": item["relevant_sources"],
                    "retrieved_sources": retrieved_sources,
                    "relevant_hits": relevant_hits,
                    "precision_at_3": round(precision, 4),
                    "latency_ms": result["latency_ms"],
                    "reader_answer": result["reader"]["answer"],
                    "reader_source": result["reader"]["source"],
                }

                method_results.append(record)

                print(
                    f"Q{item['id']:02d} | "
                    f"Precision@3: {precision:.2f} | "
                    f"Sources: {retrieved_sources}"
                )

            all_results[method] = {
                "queries": method_results,
                "mean_precision_at_3": round(total_precision / len(QUESTIONS), 4),
                "mean_latency_ms": round(
                    sum(r["latency_ms"] for r in method_results) / len(method_results),
                    2,
                ),
            }

        bm25_score = all_results["bm25"]["mean_precision_at_3"]
        dense_score = all_results["dense"]["mean_precision_at_3"]

        comparison = {
            "bm25_precision_at_3": bm25_score,
            "dense_precision_at_3": dense_score,
            "difference": round(dense_score - bm25_score, 4),
            "higher_precision": (
                "Dense"
                if dense_score > bm25_score
                else "BM25"
                if bm25_score > dense_score
                else "Tie"
            ),
        }

        evidence = {
            "assignment": "Cynaris W7D1 - Haystack Pipeline Architecture",
            "haystack_version": "3.5.0",
            "embedding_model": EMBEDDING_MODEL,
            "pdf_count": 5,
            "indexed_chunks": self.indexed_documents,
            "top_k": TOP_K,
            "evaluation_method": (
                "Source-level relevance judgments with top-K results "
                "deduplicated by PDF filename. Precision@3 uses a fixed "
                "denominator of 3. These are proxy judgments, not "
                "independent passage-level annotations."
            ),
            "comparison": comparison,
            "results": all_results,
        }

        OUTPUT_DIR.mkdir(parents=True, exist_ok=True)

        output_path = OUTPUT_DIR / "retrieval_evidence.json"
        output_path.write_text(
            json.dumps(evidence, indent=4),
            encoding="utf-8",
        )

        print("\n========== FINAL COMPARISON ==========")
        print("BM25 Precision@3:", bm25_score)
        print("Dense Precision@3:", dense_score)
        print("Higher precision:", comparison["higher_precision"])
        print("Evidence saved:", output_path)

        return evidence


def main():
    system = HaystackRetrievalSystem()

    system.index_pdfs()
    system.build_retrieval_pipelines()
    system.evaluate()


if __name__ == "__main__":
    main()

def test_reader_extracts_answer_and_source():
    reader = ExtractiveReader()

    documents = [
        Document(
            content=(
                "Haystack connects components through named inputs "
                "and outputs. Components form a modular pipeline."
            ),
            meta={"source_name": "haystack_pipeline_architecture.pdf"},
        )
    ]

    result = reader.run(
        query="How are Haystack components connected?",
        documents=documents,
    )

    assert result["answer"]
    assert result["source"] == "haystack_pipeline_architecture.pdf"
    assert result["documents"] == documents


def test_reader_handles_empty_retrieval():
    reader = ExtractiveReader()

    result = reader.run(
        query="Unknown question",
        documents=[],
    )

    assert result["answer"] == "No relevant document was retrieved."
    assert result["source"] == "None"
    assert result["documents"] == []


def test_top_k_deduplicates_repeated_pdf_sources():
    documents = [
        Document(content="Chunk 1", meta={"source_name": "a.pdf"}),
        Document(content="Chunk 2", meta={"source_name": "a.pdf"}),
        Document(content="Chunk 3", meta={"source_name": "b.pdf"}),
        Document(content="Chunk 4", meta={"source_name": "c.pdf"}),
        Document(content="Chunk 5", meta={"source_name": "d.pdf"}),
    ]

    result = HaystackRetrievalSystem.unique_source_results(documents)

    assert len(result) == TOP_K
    assert [doc.meta["source_name"] for doc in result] == ["a.pdf", "b.pdf", "c.pdf"]


def test_top_k_returns_fewer_results_when_sources_are_repeated():
    documents = [
        Document(content=f"Chunk {i}", meta={"source_name": "same.pdf"})
        for i in range(5)
    ]

    result = HaystackRetrievalSystem.unique_source_results(documents)

    assert len(result) == 1
    assert result[0] == documents[0]

    assert EVIDENCE_PATH.exists()

    evidence = json.loads(EVIDENCE_PATH.read_text(encoding="utf-8"))

    assert evidence["pdf_count"] == 5
    assert evidence["indexed_chunks"] > 0

    assert "bm25" in evidence["results"]

    evidence = json.loads(EVIDENCE_PATH.read_text(encoding="utf-8"))

    for method in ("bm25", "dense"):
        score = evidence["results"][method]["mean_precision_at_3"]
        assert 0.0 <= score <= 1.0


def test_comparison_is_consistent():
    evidence = json.loads(EVIDENCE_PATH.read_text(encoding="utf-8"))

    bm25 = evidence["results"]["bm25"]["mean_precision_at_3"]
    dense = evidence["results"]["dense"]["mean_precision_at_3"]

# W8D3 – Haystack BM25 and Dense Retrieval API Deployment

## Objective
Build a Haystack retrieval pipeline, compare BM25 and dense retrieval using 10 questions, and expose both methods through a FastAPI service prepared for Railway deployment.

## Implementation
- Indexed five PDF reference documents.
- Converted and split documents into 22 chunks.
- Implemented BM25 retrieval using InMemoryBM25Retriever.
- Implemented dense retrieval using Ollama embeddings with nomic-embed-text.
- Added an extractive reader to return answers and source documents.
- Evaluated both retrieval methods using the same 10 questions.
- Created FastAPI health and query endpoints.
- Added Dockerfile and Railway configuration.

## Evaluation Results

| Metric | BM25 | Dense |
|---|---:|---:|
| Mean Precision@3 | 0.3333 | 0.3333 |
| Mean latency (ms) | 5.63 | 55.05 |

Result: Tie in mean Precision@3.

Precision@3 uses a fixed denominator of three. Retrieved documents are deduplicated by PDF filename. Relevance judgments are source-level proxy judgments, not independent passage-level annotations.

## API Endpoints
- GET /health – reports API readiness.
- POST /query – accepts a question and retrieval method (`bm25` or `dense`).

## Validation
- Five PDFs indexed successfully.
- 22 document chunks indexed.
- Ten evaluation questions executed for each retrieval method.
- FastAPI health endpoint verified.
- BM25 and dense query endpoints verified.
- Pytest: 12 passed.
- Ruff lint: passed.
- Ruff format: passed.

## Deployment Notes
The application includes Docker and Railway configuration. Railway deployment requires a publicly reachable Ollama service configured through the OLLAMA_URL environment variable, with the nomic-embed-text model available. Localhost Ollama is only suitable for local testing.

## Limitations
- Evaluation uses five reference PDFs and ten questions.
- Source-level relevance judgments are proxies.
- Dense retrieval latency includes embedding generation.
- The Railway deployment has not yet been verified.

## Conclusion
Both retrieval methods achieved equal mean Precision@3 on the evaluation set. BM25 had lower observed latency in this local test. The FastAPI service successfully supports both retrieval methods.
